# ============================================================
# 滚雪球：大角度扩展到 40°（级1 merge 后，逐级推进，绝不跳档）
# ============================================================
# 目标：g(q) 有效范围从 ±5° 一路扩到 ±40°（工作空间 40 度）。
# 安全铁律：增益随幅度爬升，用小幅增益换算大角度 offset 会超调、逼近 guardian 70°。
#          因此每扩一档，先用上一档实测的割线增益校正，再跑下一档（数据驱动，不猜）。

# 档1：±15/20°（用 M3 实测增益起步）
python -m collect.orchestrator --open-loop --large-angle \
  --servo-port COM5 --force-port COM3 --calibration calibrations/rig2.json
# 测割线增益（找出该档最大幅度段的有效增益）
python -m model.measure_gain --session <档1会话>

# 档2：±30/40°（用档1测得的 --gain-fb/--gain-lr 校正，防超调）
python -m collect.orchestrator --open-loop --extreme \
  --servo-port COM5 --force-port COM3 --calibration calibrations/rig2.json \
  --gain-fb <档1割线增益fb*1.15> --gain-lr <档1割线增益lr*1.15>
# 再测，确认实际达到的峰值角 ≈ 目标（若显著低于 40° → 增益仍偏保守，可再放大；若接近 70° 立刻停）

# 合并重拟合 g(q)（原 M4 + 档1 + 档2 会话，覆盖 ±40°）
python -m model.fit_controller --session <M4会话> <档1会话> <档2会话> \
  --out configs/rig2_v2.json --slew-limit 2.0
# 检查：无 "⚠️ 非单调" 告警；若 ±40° 内三阶多项式折叠 → 升 --degree 或分段

# 大角度闭环 A/B（circle/waypoints ±20/30/40°，同轨迹 baseline vs 前馈）
# 判据：两轴不劣于 baseline 才 merge；折叠/外推失败 → 否决
# ⚠️ 大角度闭环已观察到震荡：根因=四缆松弛（力 ch1-ch4≈0）→ 先做共模预紧，再复测

# ============================================================
# 共模预紧标定（§8.3.2，大角度震荡第一优先，先于级1.6）
# ============================================================
# 静态持位标定已证伪：静态张力已≥T_min → c≈0，测不到动态换向松弛。
# 两种正确做法（可都跑、取较大值）：
#   A) 开环静态：发单轴恒定 offset→稳态→读实际位姿→扫正交对共模 c→拟合 c(q)
#      （u_orth=0 正是换向瞬间的差分条件，能测到位置相关的几何松弛）
#   B) 动态：扫恒定 c0，跑驱动轴正弦制造换向，测动态最小张力（覆盖速度分量）

# A) 开环静态标定（位置相关 c(q)，用雪球测得的割线增益换算，防超调）
python -m control.calibrate_co_tension --open-loop \
  --servo-port COM5 --force-port COM3 \
  --calibration calibrations/rig2.json \
  --gain-fb 0.090 --gain-lr 0.082 \
  --t-min 500 --c-max 400 --c-step 50 \
  --max-offset 500 --hold-s 4

# B) 动态标定（恒定 c0 兜底）
python -m control.calibrate_co_tension --dynamic \
  --servo-port COM5 --force-port COM3 \
  --calibration calibrations/rig2.json \
  --controller-config configs/rig2_v2.json \
  --t-min 500 --c-max 400 --c-step 50 \
  --drive-amp 20 --drive-freq 0.1 --drive-duration 20

# 检查：cat calibrations/rig2.json 确认 co_tension 已写入（A 给 [c0,c2,...]；B 给 [c0]）
# 复测大角度闭环：circle20 或 waypoints ±30/40，看震荡是否消除
# 若还有残余震荡 → 对"预紧后更硬"的系统做 Kd 扫描（Kd 已找到 U 形最优点 0.5，Kp 保持 8）
